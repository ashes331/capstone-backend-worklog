def _detect_swipe():
if len(wrist_buffer) < 10:
return None
dx = wrist_buffer[-1][0] - wrist_buffer[0][0]
dy = wrist_buffer[-1][1] - wrist_buffer[0][1]
dist = (dx**2 + dy**2) ** 0.5
if dist < 80:
return None
if abs(dx) > abs(dy):
return "swipe_right" if dx > 0 else "swipe_left"
else:
return "swipe_down" if dy > 0 else "swipe_up"
